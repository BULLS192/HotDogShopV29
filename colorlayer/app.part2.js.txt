function maskToImage(mask,rgb,w,h){const img=new ImageData(w,h);for(let p=0,i=0;p<mask.length;p++,i+=4)if(mask[p]){img.data[i]=rgb[0];img.data[i+1]=rgb[1];img.data[i+2]=rgb[2];img.data[i+3]=255}return img}
function countMask(mask){let n=0;for(const v of mask)n+=v;return n}

function traceContours(mask,w,h,tol=0){const edgeMap=new Map();const add=(x1,y1,x2,y2)=>{const k=x1+','+y1;if(!edgeMap.has(k))edgeMap.set(k,[]);edgeMap.get(k).push([x2,y2])};const at=(x,y)=>x>=0&&y>=0&&x<w&&y<h?mask[y*w+x]:0;for(let y=0;y<h;y++)for(let x=0;x<w;x++)if(at(x,y)){if(!at(x,y-1))add(x,y,x+1,y);if(!at(x+1,y))add(x+1,y,x+1,y+1);if(!at(x,y+1))add(x+1,y+1,x,y+1);if(!at(x-1,y))add(x,y+1,x,y)}const used=new Set(),loops=[];const edgeKey=(a,b)=>a[0]+','+a[1]+'>'+b[0]+','+b[1];for(const[startKey,ends]of edgeMap){const [sx,sy]=startKey.split(',').map(Number);for(const firstEnd of ends){const ek=edgeKey([sx,sy],firstEnd);if(used.has(ek))continue;let loop=[[sx,sy]],cur=[sx,sy],prev=null,next=firstEnd,guard=0;while(guard++<w*h*8){used.add(edgeKey(cur,next));loop.push(next);prev=cur;cur=next;if(cur[0]===sx&&cur[1]===sy)break;const opts=(edgeMap.get(cur[0]+','+cur[1])||[]).filter(e=>!used.has(edgeKey(cur,e)));if(!opts.length)break;next=opts.length===1?opts[0]:chooseTurn(prev,cur,opts)}if(loop.length>=4&&loop[loop.length-1][0]===sx&&loop[loop.length-1][1]===sy){loop=removeCollinear(loop);if(tol>0)loop=simplifyClosed(loop,tol);if(loop.length>=4)loops.push(loop)}}}return loops}
function chooseTurn(prev,cur,opts){const vx=cur[0]-prev[0],vy=cur[1]-prev[1];let best=opts[0],score=-99;for(const o of opts){const wx=o[0]-cur[0],wy=o[1]-cur[1];const cross=vx*wy-vy*wx,dot=vx*wx+vy*wy;const s=(cross>0?3:cross===0?2:1)+(dot<0?-2:0);if(s>score){score=s;best=o}}return best}
function removeCollinear(loop){let pts=loop.slice(0,-1),changed=true;while(changed&&pts.length>3){changed=false;const out=[];for(let i=0;i<pts.length;i++){const a=pts[(i-1+pts.length)%pts.length],b=pts[i],c=pts[(i+1)%pts.length];if((b[0]-a[0])*(c[1]-b[1])===(b[1]-a[1])*(c[0]-b[0]))changed=true;else out.push(b)}pts=out}pts.push(pts[0]);return pts}
function simplifyClosed(loop,tol){let pts=loop.slice(0,-1);if(pts.length<4)return loop;let split=0,maxD=-1;for(let i=1;i<pts.length;i++){const d=(pts[i][0]-pts[0][0])**2+(pts[i][1]-pts[0][1])**2;if(d>maxD){maxD=d;split=i}}const a=rdp(pts.slice(0,split+1),tol),b=rdp(pts.slice(split).concat([pts[0]]),tol);let out=a.slice(0,-1).concat(b.slice(0,-1));if(out.length<3)out=pts;out.push(out[0]);return out}
function rdp(points,eps){if(points.length<3)return points;const a=points[0],b=points[points.length-1];let max=0,idx=0;for(let i=1;i<points.length-1;i++){const d=pointLineDistance(points[i],a,b);if(d>max){max=d;idx=i}}if(max>eps){const l=rdp(points.slice(0,idx+1),eps),r=rdp(points.slice(idx),eps);return l.slice(0,-1).concat(r)}return[a,b]}
function pointLineDistance(p,a,b){const dx=b[0]-a[0],dy=b[1]-a[1];if(dx===0&&dy===0)return Math.hypot(p[0]-a[0],p[1]-a[1]);const t=Math.max(0,Math.min(1,((p[0]-a[0])*dx+(p[1]-a[1])*dy)/(dx*dx+dy*dy)));return Math.hypot(p[0]-(a[0]+t*dx),p[1]-(a[1]+t*dy))}
