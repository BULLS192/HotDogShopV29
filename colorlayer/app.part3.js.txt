function layerTriangles(i,z0=0){const mask=layerMask(i),{scale}=projectScale(),z1=z0+positive(els.layerMm.value,.6),tris=[];const w=state.w,h=state.h,at=(x,y)=>x>=0&&y>=0&&x<w&&y<h?mask[y*w+x]:0;for(let y=0;y<h;y++){let x=0;while(x<w){while(x<w&&!at(x,y))x++;if(x>=w)break;const x0=x;while(x<w&&at(x,y))x++;const x1=x;quad(tris,[x0*scale,y*scale,z1],[x1*scale,y*scale,z1],[x1*scale,(y+1)*scale,z1],[x0*scale,(y+1)*scale,z1]);quad(tris,[x0*scale,y*scale,z0],[x0*scale,(y+1)*scale,z0],[x1*scale,(y+1)*scale,z0],[x1*scale,y*scale,z0])}}for(let y=0;y<h;y++)for(let x=0;x<w;x++)if(at(x,y)){const x0=x*scale,x1=(x+1)*scale,y0=y*scale,y1=(y+1)*scale;if(!at(x,y-1))quad(tris,[x0,y0,z0],[x1,y0,z0],[x1,y0,z1],[x0,y0,z1]);if(!at(x+1,y))quad(tris,[x1,y0,z0],[x1,y1,z0],[x1,y1,z1],[x1,y0,z1]);if(!at(x,y+1))quad(tris,[x1,y1,z0],[x0,y1,z0],[x0,y1,z1],[x1,y1,z1]);if(!at(x-1,y))quad(tris,[x0,y1,z0],[x0,y0,z0],[x0,y0,z1],[x0,y1,z1])}return tris}

function binarySTL(tris,name='ColorLayer'){const buf=new ArrayBuffer(84+50*tris.length),dv=new DataView(buf);const hdr=new TextEncoder().encode(name.slice(0,80));new Uint8Array(buf,0,hdr.length).set(hdr);dv.setUint32(80,tris.length,true);let o=84;for(const t of tris){const n=normal(t[0],t[1],t[2]);for(const v of n){dv.setFloat32(o,v,true);o+=4}for(const p of t)for(const v of p){dv.setFloat32(o,v,true);o+=4}dv.setUint16(o,0,true);o+=2}return new Blob([buf],{type:'application/sla'})}
function normal(a,b,c){const ux=b[0]-a[0],uy=b[1]-a[1],uz=b[2]-a[2],vx=c[0]-a[0],vy=c[1]-a[1],vz=c[2]-a[2],nx=uy*vz-uz*vy,ny=uz*vx-ux*vz,nz=ux*vy-uy*vx,l=Math.hypot(nx,ny,nz)||1;return[nx/l,ny/l,nz/l]}
