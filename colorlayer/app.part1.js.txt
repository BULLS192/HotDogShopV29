function fitDims(w,h,maxDim=650){const s=Math.min(1,maxDim/Math.max(w,h));return[Math.max(1,Math.round(w*s)),Math.max(1,Math.round(h*s))]}
function loadFile(file){if(!file.type.startsWith('image/'))return;state.filenameBase=file.name.replace(/\.[^.]+$/,'');const reader=new FileReader();reader.onload=()=>loadImageDataUrl(reader.result);reader.readAsDataURL(file)}
function loadImageDataUrl(url){const img=new Image();img.onload=()=>{state.sourceImage=img;state.sourceDataUrl=url;drawSource(img);els.processBtn.disabled=false;els.status.textContent=`${img.naturalWidth} × ${img.naturalHeight}px loaded`};img.src=url}
function drawSource(img){const[w,h]=fitDims(img.naturalWidth,img.naturalHeight);state.w=w;state.h=h;for(const c of[els.sourceCanvas,els.quantCanvas]){c.width=w;c.height=h}const ctx=els.sourceCanvas.getContext('2d',{willReadFrequently:true});ctx.clearRect(0,0,w,h);ctx.drawImage(img,0,0,w,h);els.quantCanvas.getContext('2d').clearRect(0,0,w,h);state.indices=null;state.palette=[];state.backgroundIndex=null;renderEmpty();updateDimensionLine()}

async function processImage(){if(!state.sourceImage)return;els.processBtn.disabled=true;els.status.textContent='Analyzing colors…';await tick();const w=state.w,h=state.h;const work=document.createElement('canvas');work.width=w;work.height=h;const ctx=work.getContext('2d',{willReadFrequently:true});const blur=Number(els.simplify.value);if(blur)ctx.filter=`blur(${blur}px)`;ctx.drawImage(state.sourceImage,0,0,w,h);ctx.filter='none';const img=ctx.getImageData(0,0,w,h);const pal=medianCutPalette(img.data,Number(els.colorCount.value));let q=quantize(img,pal);if(Number(els.minIsland.value)>0)q=cleanupSmallIslands(q.indices,q.palette,w,h,Number(els.minIsland.value));state.indices=q.indices;state.palette=q.palette.map((rgb,i)=>({rgb,filament:'',visible:true,name:`Layer ${i+1}`}));state.backgroundIndex=guessBackground(state.indices,state.palette,w,h);renderAll();els.processBtn.disabled=false;els.exportProjectBtn.disabled=false;els.saveProjectBtn.disabled=false;els.status.textContent=`Separated into ${state.palette.length} editable colors`}
function tick(){return new Promise(r=>setTimeout(r,30))}

function medianCutPalette(data,k){const samples=[];const step=Math.max(1,Math.floor((data.length/4)/50000));for(let p=0;p<data.length;p+=4*step){if(data[p+3]<30)continue;samples.push([data[p],data[p+1],data[p+2]])}if(!samples.length)return[[255,255,255]];let boxes=[samples];while(boxes.length<k){let bi=-1,best=-1,ch=0;for(let i=0;i<boxes.length;i++){const b=boxes[i];if(b.length<2)continue;let mn=[255,255,255],mx=[0,0,0];for(const c of b)for(let j=0;j<3;j++){if(c[j]<mn[j])mn[j]=c[j];if(c[j]>mx[j])mx[j]=c[j]}const ranges=mx.map((v,j)=>v-mn[j]);const c=ranges.indexOf(Math.max(...ranges));const score=Math.max(...ranges)*b.length;if(score>best){best=score;bi=i;ch=c}}if(bi<0)break;const box=boxes.splice(bi,1)[0];box.sort((a,b)=>a[ch]-b[ch]);const mid=Math.floor(box.length/2);boxes.push(box.slice(0,mid),box.slice(mid))}return boxes.map(b=>{const s=b.reduce((a,c)=>[a[0]+c[0],a[1]+c[1],a[2]+c[2]],[0,0,0]);return s.map(v=>Math.round(v/Math.max(1,b.length)))})}
function quantize(img,palette){const d=img.data,indices=new Uint16Array(img.width*img.height),counts=new Array(palette.length).fill(0);for(let i=0,p=0;i<d.length;i+=4,p++){if(d[i+3]<15){indices[p]=65535;continue}let best=0,dist=Infinity;for(let j=0;j<palette.length;j++){const c=palette[j],dr=d[i]-c[0],dg=d[i+1]-c[1],db=d[i+2]-c[2],dd=dr*dr+dg*dg+db*db;if(dd<dist){dist=dd;best=j}}indices[p]=best;counts[best]++}const order=counts.map((n,i)=>[n,i]).filter(x=>x[0]>0).sort((a,b)=>b[0]-a[0]).map(x=>x[1]);const remap=new Map(order.map((old,nu)=>[old,nu]));for(let p=0;p<indices.length;p++)if(indices[p]!==65535)indices[p]=remap.get(indices[p]);return{indices,palette:order.map(i=>palette[i])}}
function cleanupSmallIslands(indices,palette,w,h,minSize){const out=new Uint16Array(indices);const seen=new Uint8Array(indices.length),dirs=[[1,0],[-1,0],[0,1],[0,-1]];for(let p=0;p<out.length;p++){if(seen[p]||out[p]===65535)continue;const target=out[p],queue=[p],comp=[],neighbors=new Map();seen[p]=1;for(let qi=0;qi<queue.length;qi++){const cur=queue[qi],x=cur%w,y=Math.floor(cur/w);comp.push(cur);for(const[dx,dy]of dirs){const nx=x+dx,ny=y+dy;if(nx<0||ny<0||nx>=w||ny>=h)continue;const np=ny*w+nx,v=out[np];if(v===target&&!seen[np]){seen[np]=1;queue.push(np)}else if(v!==target&&v!==65535)neighbors.set(v,(neighbors.get(v)||0)+1)}}if(comp.length<minSize&&neighbors.size){let replacement=[...neighbors.entries()].sort((a,b)=>b[1]-a[1])[0][0];for(const cp of comp)out[cp]=replacement}}const counts=new Array(palette.length).fill(0);for(const v of out)if(v!==65535)counts[v]++;const order=counts.map((n,i)=>[n,i]).filter(x=>x[0]>0).sort((a,b)=>b[0]-a[0]).map(x=>x[1]);const map=new Map(order.map((old,nu)=>[old,nu]));for(let i=0;i<out.length;i++)if(out[i]!==65535)out[i]=map.get(out[i]);return{indices:out,palette:order.map(i=>palette[i])}}
function guessBackground(indices,palette,w,h){const counts=new Map();for(let x=0;x<w;x++){for(const p of[x,(h-1)*w+x])if(indices[p]!==65535)counts.set(indices[p],(counts.get(indices[p])||0)+1)}for(let y=1;y<h-1;y++){for(const p of[y*w,y*w+w-1])if(indices[p]!==65535)counts.set(indices[p],(counts.get(indices[p])||0)+1)}return counts.size?[...counts.entries()].sort((a,b)=>b[1]-a[1])[0][0]:null}
